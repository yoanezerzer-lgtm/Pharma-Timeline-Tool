// Function to open PDF with Chrome browser preference
function openPdfWithChrome(url) {
try {
  // For preview mode with local files, try to open with Chrome
  if (pdfBaseUrl && pdfBaseUrl.indexOf('file://') === 0) {
    // Create a temporary link element
    var link = document.createElement('a');
    link.href = url;
    link.target = '_blank';
    link.rel = 'noopener noreferrer';
    
    // Try to detect Chrome and use it preferentially
    var isChrome = navigator.userAgent.indexOf('Chrome') > -1 && navigator.vendor.indexOf('Google Inc') > -1;
    
    if (isChrome) {
      // Chrome detected, open normally
      link.click();
    } else {
      // Not Chrome, try to suggest Chrome or open with default
      if (confirm('For best PDF viewing experience, we recommend using Chrome browser. Open with default browser?')) {
        link.click();
      }
    }
  } else {
    // Production mode, open normally
    window.open(url, '_blank', 'noopener,noreferrer');
  }
} catch (e) {
  console.error('Error opening PDF:', e);
  // Fallback to normal link opening
  window.open(url, '_blank', 'noopener,noreferrer');
}
}
